def decide(key: str, state, questions: dict, *, attempts: int = 5, timeout_s: float = 60.0,
           base_delay_s: float = 1.0, max_delay_s: float = 30.0) -> dict:
    """One Decisions request with bounded retries on 429/5xx/network errors."""
    data = json.dumps({"model": MODEL, "state": state, "questions": questions}).encode()
    headers = {"Authorization": f"Bearer {key}", "Content-Type": "application/json"}
    for attempt in range(attempts):
        last = attempt == attempts - 1
        try:
            req = urllib.request.Request(URL, data=data, headers=headers, method="POST")
            with _OPENER.open(req, timeout=timeout_s) as resp:
                return json.load(resp)
        except urllib.error.HTTPError as e:
            detail = _error_detail(e)
            if not (e.code == 429 or e.code >= 500) or last:
                raise SnapError(e.code, detail.get("error_type", "http_error"), detail.get("message", "")) from e
            retry_after = e.headers.get("Retry-After") if e.headers else None
            if retry_after and retry_after.isdigit():
                time.sleep(min(max_delay_s, float(retry_after)))
                continue
        except (OSError, http.client.HTTPException, ValueError) as e:
            # Timeouts, resets, truncated/invalid bodies: outcome unknown; retrying is safe (no side effects).
            if last:
                raise SnapError(None, "network", f"{type(e).__name__}: {e}") from e
        time.sleep(random.uniform(0, min(max_delay_s, base_delay_s * 2 ** attempt)))
    raise SnapError(None, "internal", "retry loop ended without a result")
