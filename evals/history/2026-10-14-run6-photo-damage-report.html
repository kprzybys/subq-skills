"""Check return photos for product damage with SubQ Snap (POST /v1/systemone).

Usage:
    SUBQ_API_KEY=... python3 check_photos.py RETURN_DIR [RETURN_DIR ...]

Each RETURN_DIR holds the files for one return: up to 3 JPEG/PNG photos, or a short video
(mp4/mov/webm/mkv/avi/m4v) when there are no photos. For a video, 4 frames are sampled with
ffmpeg (must be on PATH), because the API does not accept video.

Output: one JSON line per return on stdout, with the raw probabilities kept so thresholds can be
changed later without calling the API again:
    {"return": ..., "status": "damaged" | "not_damaged" | "needs_review" | "error",
     "damage_types": [...], "reasons": [...], "probabilities": {...}, "input_tokens": ...}
"needs_review" and "error" both mean a person must look at the return.

Error handling:
- Retried with exponential backoff and full jitter (bounded attempts): HTTP 429 (rate limit),
  HTTP 5xx (503 = temporarily unavailable), timeouts and dropped connections. A decision request
  has no side effects, so resending one whose outcome is unknown is safe.
- Not retried, that return goes to a person: HTTP 400 (input rejected or too large) and
  422 (invalid request or image). Resending the same request would give the same answer.
- Not retried, the whole run stops: HTTP 401 (bad key), 402 (no credits), 403 (key cannot use
  the model), and any redirect (never followed, so the key is never sent to another host).
  These would fail every remaining return the same way.

Exit codes: 0 all returns decided automatically, 1 at least one return needs a person or failed,
2 the run was stopped by an auth/credit error or bad usage.

The thresholds below are placeholders: calibrate them on returns whose outcome you already know.
"""

BASE_DELAY_S = 1.0

VIDEO_FRAMES = 4
MAX_IMAGE_BYTES = 4 * 1024 * 1024  # API limit per decoded image
MAX_TOTAL_BYTES = 8 * 1024 * 1024  # API limit for all images together
MAX_SIDE_PX = 1600  # images larger than this are shrunk when Pillow is installed

# Placeholder thresholds; tune on labelled returns.
CLEAR_YES = 0.80  # noul at or above: treat as yes
CLEAR_NO = 0.20  # noul at or below: treat as no; anything between goes to a person
JUDGEABLE_MIN = 0.70  # below: photos do not show the product well enough to decide

PHOTO_EXTS = {".jpg", ".jpeg", ".png"}
VIDEO_EXTS = {".mp4", ".mov", ".m4v", ".webm", ".mkv", ".avi"}

DAMAGE_TYPES = {

    "dented": "The product has a dent, bend, crush mark or other deformation of its shape.",
    "missing_part": "A part, component or accessory that belongs to the product is visibly missing or "

QUESTIONS: dict = {

class SnapError(Exception):
    """No decision was produced. `status` is the HTTP status, or None for a network failure."""

    def __init__(self, status: int | None, error_type: str, message: str) -> None:
        super().__init__(f"{status} {error_type}: {message}")
        self.status = status
        self.error_type = error_type


class FatalSnapError(SnapError):
    """Auth, credit or redirect problem: every further request would fail too."""


class _NoRedirect(urllib.request.HTTPRedirectHandler):
    def redirect_request(self, req, fp, code, msg, headers, newurl):  # noqa: ANN001
        return None  # urllib raises HTTPError for the 3xx, so the key is not resent elsewhere


_OPENER = urllib.request.build_opener(_NoRedirect)


def _error_detail(e: urllib.error.HTTPError) -> dict:
    try:
        payload = json.load(e)
    except (ValueError, OSError, http.client.HTTPException):
        return {"error_type": "http_error", "message": ""}
    detail = payload.get("detail") if isinstance(payload, dict) else None
    return detail if isinstance(detail, dict) else {"error_type": "http_error", "message": str(payload)[:200]}


def decide(key: str, state: dict, questions: dict, images: list[dict]) -> dict:
    body = json.dumps({"model": MODEL, "state": state, "questions": questions, "images": images}).encode()
    headers = {"Authorization": f"Bearer {key}", "Content-Type": "application/json"}
    for attempt in range(ATTEMPTS):
        last = attempt == ATTEMPTS - 1
        try:
            request = urllib.request.Request(URL, data=body, headers=headers)
            with _OPENER.open(request, timeout=TIMEOUT_S) as response:
                return json.load(response)
        except urllib.error.HTTPError as e:
            d = _error_detail(e)
            err_type, msg = d.get("error_type", "http_error"), d.get("message", "")
            if e.code in (401, 402, 403) or 300 <= e.code < 400:
                raise FatalSnapError(e.code, err_type, msg) from e
            if not (e.code == 429 or e.code >= 500) or last:
                raise SnapError(e.code, err_type, msg) from e
        except (OSError, http.client.HTTPException, ValueError) as e:
            # URLError, timeouts, resets, truncated or non-JSON bodies: outcome unknown, safe to retry.
            if last:
                raise SnapError(None, "network", f"{type(e).__name__}: {e}") from e
        time.sleep(random.uniform(0, min(MAX_DELAY_S, BASE_DELAY_S * 2**attempt)))
    raise SnapError(None, "internal", "retry loop ended without a result")


# --------------------------------------------------------------------------- inputs


class InputProblem(Exception):
    """The return's files cannot be sent as-is; a person should look at them."""


def _media_type(data: bytes) -> str | None:
    if data.startswith(b"\xff\xd8\xff"):
        return "image/jpeg"
    if data.startswith(b"\x89PNG\r\n\x1a\n"):
        return "image/png"
    return None


def _shrink(data: bytes) -> bytes:
    """Re-encode as JPEG no larger than MAX_SIDE_PX; needs Pillow."""
    try:
        from PIL import Image, ImageOps
    except ImportError as e:
        raise InputProblem("image over 4 MiB and Pillow is not installed to shrink it") from e
    with Image.open(io.BytesIO(data)) as img:
        img = ImageOps.exif_transpose(img).convert("RGB")
        img.thumbnail((MAX_SIDE_PX, MAX_SIDE_PX))
        out = io.BytesIO()
        img.save(out, format="JPEG", quality=85)
        return out.getvalue()


def _image_entry(data: bytes, label: str) -> dict:
    media = _media_type(data)
    if media is None:
        raise InputProblem(f"{label} is not a JPEG or PNG")
    if len(data) > MAX_IMAGE_BYTES:
        data, media = _shrink(data), "image/jpeg"
        if len(data) > MAX_IMAGE_BYTES:
            raise InputProblem(f"{label} is still over 4 MiB after shrinking")
    return {"mediaType": media, "base64": base64.b64encode(data).decode("ascii"), "_size": len(data)}


def _run(cmd: list[str], timeout: float) -> bytes:
    try:
        return subprocess.run(cmd, capture_output=True, check=True, timeout=timeout).stdout
    except (subprocess.CalledProcessError, subprocess.TimeoutExpired, OSError) as e:
        raise InputProblem(f"could not read video: {type(e).__name__}") from e


def _video_frames(path: Path) -> list[bytes]:
    if not (shutil.which("ffmpeg") and shutil.which("ffprobe")):
        raise InputProblem("video received but ffmpeg/ffprobe is not installed")
    src = f"file:{path.resolve()}"  # "file:" stops ffmpeg from treating the name as a protocol or option
    out = _run(["ffprobe", "-v", "error", "-show_entries", "format=duration", "-of", "csv=p=0", src], 30)
    try:
        duration = float(out.strip())
    except ValueError as e:
        raise InputProblem("could not read video duration") from e
    if duration <= 0:
        raise InputProblem("video has no duration")
    frames = []
    for i in range(1, VIDEO_FRAMES + 1):
        t = duration * i / (VIDEO_FRAMES + 1)  # evenly spaced, skipping the very start and end
        frame = _run(
            ["ffmpeg", "-v", "error", "-ss", f"{t:.3f}", "-i", src, "-frames:v", "1",
             "-vf", f"scale='min({MAX_SIDE_PX},iw)':-2", "-q:v", "3", "-f", "image2", "-c:v", "mjpeg", "pipe:1"],
            60,
        )
        if frame:
            frames.append(frame)
    if not frames:
        raise InputProblem("no frames could be extracted from the video")
    return frames


def load_images(folder: Path) -> tuple[list[dict], str]:
    """Return (API image entries, source description). Raises InputProblem for a person to handle."""
    files = sorted(p for p in folder.iterdir() if p.is_file() and not p.name.startswith("."))
    photos = [p for p in files if p.suffix.lower() in PHOTO_EXTS]
    videos = [p for p in files if p.suffix.lower() in VIDEO_EXTS]

    if photos:
        if len(photos) > MAX_PHOTOS:
            raise InputProblem(f"{len(photos)} photos, more than the {MAX_PHOTOS} allowed")
        images = [_image_entry(p.read_bytes(), p.name) for p in photos]
        source = f"{len(images)} customer photo(s)" + (" (a video was also attached and ignored)" if videos else "")
    elif videos:
        if len(videos) > 1:
            raise InputProblem(f"{len(videos)} videos attached, expected one")
        images = [_image_entry(f, f"{videos[0].name} frame {i}") for i, f in enumerate(_video_frames(videos[0]), 1)]
        source = f"{len(images)} frames sampled evenly from a customer video"
    else:
        raise InputProblem("no JPEG/PNG photos or video found")

    if sum(img["_size"] for img in images) > MAX_TOTAL_BYTES:
        raise InputProblem("images together exceed 8 MiB")
    for img in images:
        del img["_size"]
    return images, source


# --------------------------------------------------------------------------- decision


def classify(answers: dict) -> tuple[str, list[str], list[str], dict]:
    """Turn answers into (status, damage_types, reasons, probabilities). Code owns these rules."""
    probs = {name: answers[name]["noul"] for name in QUESTIONS}
    reasons = []

    if probs["judgeable"] < JUDGEABLE_MIN:
        return "needs_review", [], ["photos may not show the product clearly"], probs

    damaged = probs["damaged"]
    types = [t for t in DAMAGE_TYPES if probs[t] >= CLEAR_YES]
    unsure_types = [t for t in DAMAGE_TYPES if CLEAR_NO < probs[t] < CLEAR_YES]

    if CLEAR_NO < damaged < CLEAR_YES:
        reasons.append(f"unsure whether the product is damaged (p={damaged:.2f})")
    elif damaged >= CLEAR_YES:
        if not types:
            reasons.append("damage seen but no clear damage type")
        if unsure_types:
            reasons.append("unsure about damage type(s): " + ", ".join(unsure_types))
    elif types:  # damaged is a clear no, yet some type is a clear yes
        reasons.append("conflicting answers: no damage overall but " + ", ".join(types))

    if reasons:
        return "needs_review", types, reasons, probs
    return ("damaged" if damaged >= CLEAR_YES else "not_damaged"), types, [], probs


def check_return(key: str, folder: Path) -> dict:
    result: dict = {"return": folder.name, "path": str(folder)}
    try:
        images, source = load_images(folder)
    except (InputProblem, OSError) as e:
        return {**result, "status": "needs_review", "damage_types": [], "reasons": [str(e)]}

    state = {
        "task": "Inspect a product a customer is returning and judge its physical condition.",
        "images": source,
        "note": "All images show the same returned item; judge the product, not the packaging or background.",
    }
    try:
        response = decide(key, state, QUESTIONS, images)
    except FatalSnapError:
        raise
    except SnapError as e:
        return {**result, "status": "error", "damage_types": [], "reasons": [f"API error, not decided: {e}"]}

    try:
        status, types, reasons, probs = classify(response["answers"])
    except (KeyError, TypeError) as e:
        return {**result, "status": "error", "damage_types": [], "reasons": [f"unexpected response shape: {e!r}"]}
    return {
        **result,
        "status": status,
        "damage_types": types,
        "reasons": reasons,
        "source": source,
        "probabilities": probs,
        "input_tokens": response.get("usage", {}).get("input_tokens"),
    }


def main(argv: list[str] | None = None) -> int:
    parser = argparse.ArgumentParser(description="Check customer return photos for damage with SubQ Snap.")
    parser.add_argument("returns", nargs="+", type=Path, help="one folder per return")
    args = parser.parse_args(argv)

    key = os.environ.get("SUBQ_API_KEY")
    if not key:
        print("check_photos: SUBQ_API_KEY is not set", file=sys.stderr)
        return 2

    needs_person = False
    for folder in args.returns:
        if not folder.is_dir():
            row = {"return": folder.name, "path": str(folder), "status": "needs_review",
                   "damage_types": [], "reasons": ["not a folder"]}
        else:
            try:
                row = check_return(key, folder)
            except FatalSnapError as e:
                print(f"check_photos: stopping, {e}", file=sys.stderr)
                return 2
        needs_person |= row["status"] in ("needs_review", "error")
        print(json.dumps(row, ensure_ascii=False), flush=True)
    return 1 if needs_person else 0


if __name__ == "__main__":
    sys.exit(main())

"""Classify product-return photos for damage with the SubQ Decisions API.

Usage:
    export SUBQ_API_KEY=...
    python check_photos.py RETURN_DIR [RETURN_DIR ...]
    python check_photos.py returns/*/          # one folder per return

Each RETURN_DIR holds the customer's media for one return: up to 3 JPEG/PNG
photos, or a short video (frames are sampled with ffmpeg, because the API
does not accept video). Writes one JSON object per return to stdout:

    {"return_id": "...", "decision": "damaged" | "not_damaged" | "needs_review",
     "damage_type": "cracked" | "scratched" | "dented" | "missing_part" | null,
     "reasons": [...], "scores": {...}}

Anything uncertain, unreadable, or that failed at the API ends up as
"needs_review" so a person looks at it.

Requires: requests, Pillow; ffmpeg/ffprobe on PATH for video returns.

def encode_image(img):
    """Normalise a PIL image to an API-ready JPEG under the size limits."""
    img = ImageOps.exif_transpose(img)
    if img.mode != "RGB":
        img = img.convert("RGB")
    img.thumbnail((MAX_SIDE_PX, MAX_SIDE_PX))
    for quality in (90, 80, 70, 60, 50):
        buf = io.BytesIO()
        img.save(buf, format="JPEG", quality=quality)
        if buf.tell() <= MAX_IMAGE_BYTES:
            break
    else:
        img.thumbnail((MAX_SIDE_PX // 2, MAX_SIDE_PX // 2))
        buf = io.BytesIO()
        img.save(buf, format="JPEG", quality=70)
    return {"mediaType": "image/jpeg",
            "base64": base64.b64encode(buf.getvalue()).decode("ascii")}


def load_photo(path):
    with Image.open(path) as img:
        img.load()
        return encode_image(img)


def sample_video_frames(path, n=VIDEO_FRAMES):
    """Grab n evenly spaced frames from a video with ffmpeg."""
    if not shutil.which("ffmpeg") or not shutil.which("ffprobe"):
        raise RuntimeError("ffmpeg/ffprobe not installed; cannot read video")
    probe = subprocess.run(
        ["ffprobe", "-v", "error", "-show_entries", "format=duration",
         "-of", "default=noprint_wrappers=1:nokey=1", str(path)],
        capture_output=True, text=True, timeout=60)
    try:
        duration = float(probe.stdout.strip())
    except ValueError:
        raise RuntimeError(f"cannot read video duration: {probe.stderr.strip()[:200]}")
    if duration <= 0:
        raise RuntimeError("video has zero duration")

    frames = []
    with tempfile.TemporaryDirectory() as tmp:
        for i in range(n):
            t = duration * (2 * i + 1) / (2 * n)   # midpoints of n equal slices
            out = Path(tmp) / f"frame{i}.png"
            subprocess.run(
                ["ffmpeg", "-v", "error", "-y", "-ss", f"{t:.3f}", "-i", str(path),
                 "-frames:v", "1", str(out)],
                capture_output=True, timeout=120)
            if out.exists():
                with Image.open(out) as img:
                    img.load()
                    frames.append(encode_image(img))
    if not frames:
        raise RuntimeError("could not extract any frames from video")
    return frames


def collect_media(folder):
    """Return (images, source_description, notes) for one return folder."""
    files = sorted(p for p in folder.iterdir() if p.is_file() and not p.name.startswith("."))
    photos = [p for p in files if p.suffix.lower() in PHOTO_EXTS]
    videos = [p for p in files if p.suffix.lower() in VIDEO_EXTS]
    ignored = [p.name for p in files if p not in photos and p not in videos]

    notes = []
    if ignored:
        notes.append(f"ignored unsupported files: {', '.join(ignored)}")

    if photos:
        if len(photos) > MAX_PHOTOS:
            notes.append(f"{len(photos)} photos found, expected at most {MAX_PHOTOS}; "
                         f"using the first {MAX_PHOTOS}")
            photos = photos[:MAX_PHOTOS]
        if videos:
            notes.append("photos and video both present; used photos only")
        images = []
        for p in photos:
            try:
                images.append(load_photo(p))
            except (UnidentifiedImageError, OSError) as e:
                notes.append(f"unreadable photo {p.name}: {e}")
        return images, f"{len(images)} customer photo(s)", notes

    if videos:
        if len(videos) > 1:
            notes.append(f"{len(videos)} videos found; used {videos[0].name}")
        try:
            frames = sample_video_frames(videos[0])
        except (RuntimeError, OSError, subprocess.SubprocessError, UnidentifiedImageError) as e:
            notes.append(f"video {videos[0].name}: {e}")
            return [], "video", notes
        return frames, f"{len(frames)} frames sampled in time order from a customer video", notes

    return [], "nothing", notes

def retry_after_seconds(resp):
    value = resp.headers.get("Retry-After") if resp is not None else None
    if value is None:
        return None
    try:
        return max(0.0, float(value))
    except ValueError:
        return None


def error_detail(resp):
    try:
        detail = resp.json().get("detail", {})
        if isinstance(detail, dict):
            return f"{detail.get('error_type', '')}: {detail.get('message', '')}".strip(": ")
        return str(detail)
    except ValueError:
        return resp.text[:300]


def call_api(session, payload, api_key, max_attempts, timeout):
    """POST to the Decisions API.

    Retried (exponential backoff with jitter, honouring Retry-After):
      - network errors: connection failures, resets, timeouts
      - HTTP 408, 429, 500, 502, 503, 504
    Not retried:
      - 401, 402, 403: bad key, no credits, model not allowed. Raised as
        FatalApiError so the run stops instead of failing every return.
      - 400 (over token limit / rejected), 422 (invalid request or image),
        and any other 4xx: the same request would fail again.
      - 2xx responses that are not the expected JSON (treated as a bad answer
        and sent to review rather than re-billed).
    """
    headers = {"Authorization": f"Bearer {api_key}", "Content-Type": "application/json"}
    last_error = None
    for attempt in range(1, max_attempts + 1):
        resp = None
        try:
            resp = session.post(API_URL, headers=headers, json=payload, timeout=timeout)
        except (requests.ConnectionError, requests.Timeout) as e:
            last_error = ApiError(f"network error: {e.__class__.__name__}: {e}", retryable=True)
        else:
            if resp.ok:
                try:
                    return resp.json()
                except ValueError:
                    raise ApiError(f"non-JSON response (HTTP {resp.status_code})")
            msg = f"HTTP {resp.status_code}: {error_detail(resp)}"
            if resp.status_code in FATAL_STATUS:
                raise FatalApiError(msg, status=resp.status_code)
            if resp.status_code not in RETRYABLE_STATUS:
                raise ApiError(msg, status=resp.status_code)
            last_error = ApiError(msg, status=resp.status_code, retryable=True)

        if attempt == max_attempts:
            break
        delay = retry_after_seconds(resp)
        if delay is None:
            delay = random.uniform(0, min(30.0, 1.0 * 2 ** attempt))
        delay = min(delay, 120.0)
        print(f"  retry {attempt}/{max_attempts - 1} in {delay:.1f}s after {last_error}",
              file=sys.stderr)
        time.sleep(delay)

    raise ApiError(f"gave up after {max_attempts} attempts; last error: {last_error}",
                   status=last_error.status if last_error else None)

def decide(answers, args):
    """Turn API probabilities into damaged / not_damaged / needs_review."""
    try:
        p_assessable = float(answers["assessable"]["noul"])
        p_damaged = float(answers["damaged"]["noul"])
        dt = answers["damage_type"]
        probs = {k: float(v) for k, v in dt.get("probabilities", {}).items()}
        top = dt.get("choice") or max(probs, key=probs.get)
        top_p = probs.get(top, float(dt.get("confidence", 0.0)))
    except (KeyError, TypeError, ValueError) as e:
        return "needs_review", None, [f"unexpected API response shape: {e!r}"], {}

    scores = {"p_assessable": round(p_assessable, 3), "p_damaged": round(p_damaged, 3),
              "damage_type_probabilities": {k: round(v, 3) for k, v in probs.items()}}
    reasons = []

    if p_assessable < args.assessable_threshold:
        reasons.append(f"images may not show the product clearly (p={p_assessable:.2f})")
        return "needs_review", None, reasons, scores

    if p_damaged <= args.not_damaged_threshold:
        return "not_damaged", None, reasons, scores
    if p_damaged < args.damaged_threshold:
        reasons.append(f"damage uncertain (p_damaged={p_damaged:.2f})")
        return "needs_review", None, reasons, scores

    if top == "other":
        reasons.append("damaged, but damage does not fit a known category")
        return "needs_review", None, reasons, scores
    if top_p < args.type_threshold:
        reasons.append(f"damaged, but damage type uncertain ({top} p={top_p:.2f})")
        return "needs_review", top, reasons, scores
    return "damaged", top, reasons, scores


def check_return(session, folder, api_key, args):
    result = {"return_id": folder.name, "decision": "needs_review",
              "damage_type": None, "reasons": [], "scores": {}}
    images, source, notes = collect_media(folder)
    result["reasons"].extend(notes)
    if not images:
        result["reasons"].append(f"no usable images (found: {source})")
        return result
    images = images[:MAX_API_IMAGES]

    try:
        response = call_api(session, build_request(images, source), api_key,
                            args.max_attempts, args.timeout)
    except FatalApiError:
        raise
    except ApiError as e:
        result["reasons"].append(f"API error: {e}")
        return result

    decision, damage_type, reasons, scores = decide(response.get("answers", {}), args)
    result.update(decision=decision, damage_type=damage_type, scores=scores)
    result["reasons"].extend(reasons)
    # Anything odd about the input (unreadable photo, extra files) means a person checks.
    if notes and decision != "needs_review" and any(n.startswith("unreadable") for n in notes):
        result["decision"] = "needs_review"
        result["reasons"].append("some photos were unreadable; decision based on partial input")
    return result
